namespace = mages_commune_events_2


# Mage's Commune versions of the vanilla estate events. Each fires only when the vanilla event would not.


scripted_effect mages_commune_find_coup_interlocutor_effect = {
	random_list = {
		1 = {
			trigger = { exists = cp:councillor_chancellor }
			cp:councillor_chancellor = { save_scope_as = interlocutor }
		}
		1 = {
			trigger = { exists = court_position:chief_eunuch_court_position }
			court_position:chief_eunuch_court_position = { save_scope_as = interlocutor }
		}
		1 = {
			trigger = { exists = cp:councillor_steward }
			cp:councillor_steward = { save_scope_as = interlocutor }
		}
	}
	if = {
		limit = { NOT = { exists = scope:interlocutor } }
		ordered_courtier = {
			order_by = intrigue
			save_scope_as = interlocutor
		}
	}
	if = {
		limit = { NOT = { exists = scope:interlocutor } }
		ordered_pool_character = {
			province = root.location
			order_by = intrigue
			save_scope_as = interlocutor
		}
	}
}

# Mage's Commune version of grand_ambitions.0120
mages_commune_events_2.1 = {
	type = character_event
	title = mages_commune_events_2.1.t
	desc = mages_commune_events_2.1.desc
	theme = administrative
	override_background = { reference = estate }

	left_portrait = {
		character = root
		animation = survey
	}
	right_portrait = {
		character = scope:interlocutor
		animation = debating
	}

	cooldown = { years = 10 }

	trigger = {
		# Only where the vanilla event would not fire
		NOT = {
		has_domicile = yes
		domicile = {
			is_domicile_type = estate
			num_domicile_buildings < 6
		}
		OR = {
			number_maa_regiments_of_base_type:skirmishers > 0
			number_maa_regiments_of_base_type:heavy_infantry > 0
			number_maa_regiments_of_base_type:pikemen > 0
			number_maa_regiments_of_base_type:light_cavalry > 0
			number_maa_regiments_of_base_type:heavy_cavalry > 0
			number_maa_regiments_of_base_type:archers > 0
		}
	}

		has_domicile = yes
		domicile = {
			is_domicile_type = mages_commune
			num_domicile_buildings < 6
		}
		OR = {
			number_maa_regiments_of_base_type:skirmishers > 0
			number_maa_regiments_of_base_type:heavy_infantry > 0
			number_maa_regiments_of_base_type:pikemen > 0
			number_maa_regiments_of_base_type:light_cavalry > 0
			number_maa_regiments_of_base_type:heavy_cavalry > 0
			number_maa_regiments_of_base_type:archers > 0
		}
	}

	immediate = {
		mages_commune_find_coup_interlocutor_effect = yes
		domicile = { save_scope_as = estate }
		# Pick the best building for the MaA type they have the most of
		random_list = {
			0 = { # Barracks for Skirmishers, Heavy Infantry, and Spearmen
				trigger = {
					NOT = {
						domicile = { has_domicile_building = mages_commune_barracks_06 }
					}
				}
				modifier = {
					add = number_maa_regiments_of_base_type:skirmishers
					add = number_maa_regiments_of_base_type:heavy_infantry
					add = number_maa_regiments_of_base_type:pikemen
				}
				save_scope_value_as = {
					name = estate_building
					value = flag:barracks
				}
				random_maa_regiment = {
					limit = {
						OR = {
							is_unit_type = skirmishers
							is_unit_type = heavy_infantry
							is_unit_type = pikemen
						}
					}
					save_scope_as = maa_flavor
				}
			}
			0 = { # Pasteurs for Cavalry
				trigger = {
					NOT = {
						domicile = { has_domicile_building = mages_commune_grazing_land_06 }
					}
				}
				modifier = {
					add = number_maa_regiments_of_base_type:light_cavalry
					add = number_maa_regiments_of_base_type:heavy_cavalry
				}
				save_scope_value_as = {
					name = estate_building
					value = flag:grazing_land
				}
				random_maa_regiment = {
					limit = {
						OR = {
							is_unit_type = light_cavalry
							is_unit_type = heavy_cavalry
						}
					}
					save_scope_as = maa_flavor
				}
			}
			0 = { # Watchtower for Archers
				trigger = {
					NOT = {
						domicile = { has_domicile_building = mages_commune_watchtower_06 }
					}
				}
				modifier = {
					add = number_maa_regiments_of_base_type:archers
				}
				save_scope_value_as = {
					name = estate_building
					value = flag:watchtower
				}
				random_maa_regiment = {
					limit = { is_unit_type = archers }
					save_scope_as = maa_flavor
				}
			}
		}
	}

	option = { # Construct Barracks
		name = mages_commune_events_2.1.a
		trigger = {
			scope:estate_building ?= flag:barracks
			number_maa_regiments_of_base_type:pikemen >=1
		}
		# Build or upgrade barracks building
		domicile = {
			if = {
				limit = { has_domicile_building_or_higher = mages_commune_barracks_01 }
				switch = {
					trigger = has_domicile_building
					mages_commune_barracks_01 = { add_domicile_building = mages_commune_barracks_02 }
					mages_commune_barracks_02 = { add_domicile_building = mages_commune_barracks_03 }
					mages_commune_barracks_03 = { add_domicile_building = mages_commune_barracks_04 }
					mages_commune_barracks_04 = { add_domicile_building = mages_commune_barracks_05 }
					mages_commune_barracks_05 = { add_domicile_building = mages_commune_barracks_06 }
				}
			}
			else = {
				add_domicile_building = mages_commune_barracks_01
			}
		}
	}

	option = { # Construct Pasteur
		name = mages_commune_events_2.1.b
		trigger = { scope:estate_building ?= flag:grazing_land }
		# Build or upgrade pasteurs building
		domicile = {
			if = {
				limit = { has_domicile_building_or_higher = mages_commune_grazing_land_01 }
				switch = {
					trigger = has_domicile_building
					mages_commune_grazing_land_01 = { add_domicile_building = mages_commune_grazing_land_02 }
					mages_commune_grazing_land_02 = { add_domicile_building = mages_commune_grazing_land_03 }
					mages_commune_grazing_land_03 = { add_domicile_building = mages_commune_grazing_land_04 }
					mages_commune_grazing_land_04 = { add_domicile_building = mages_commune_grazing_land_05 }
					mages_commune_grazing_land_05 = { add_domicile_building = mages_commune_grazing_land_06 }
				}
			}
			else = {
				add_domicile_building = mages_commune_grazing_land_01
			}
		}
	}

	option = { # Construct Watchtower
		name = mages_commune_events_2.1.c
		trigger = { scope:estate_building ?= flag:watchtower }
		# Build or upgrade watchtower building
		domicile = {
			if = {
				limit = { has_domicile_building_or_higher = mages_commune_watchtower_01 }
				switch = {
					trigger = has_domicile_building
					mages_commune_watchtower_01 = { add_domicile_building = mages_commune_watchtower_02 }
					mages_commune_watchtower_02 = { add_domicile_building = mages_commune_watchtower_03 }
					mages_commune_watchtower_03 = { add_domicile_building = mages_commune_watchtower_04 }
					mages_commune_watchtower_04 = { add_domicile_building = mages_commune_watchtower_05 }
					mages_commune_watchtower_05 = { add_domicile_building = mages_commune_watchtower_06 }
				}
			}
			else = {
				add_domicile_building = mages_commune_watchtower_01
			}
		}
	}

	option = { # Fallback Estate Upgrade
		name = mages_commune_events_2.1.d
		remove_short_term_gold = medium_gold_value
		add_character_modifier = {
			modifier = ep3_grand_ambitions_local_labor_modifier
			years = 10
		}
	}
}

mages_commune_events_2.2 = {
	type = character_event
	title = mages_commune_events_2.2.t
	desc = mages_commune_events_2.2.desc
	theme = administrative
	left_portrait = {
		character = root
		animation = thinking
	}
	right_portrait = {
		character = scope:other_family
		animation = personality_honorable
	}
	cooldown = { years = 15 }

	trigger = {
		# Only where the vanilla event would not fire
		NOT = {
		government_has_mechanic = administrative
		government_allows = noble_families
		is_available = yes
		house ?= {
			is_powerful_family = yes
		}
		domicile ?= {
			is_domicile_type = estate
			trigger_if = {
				limit = {
					root = { top_liege != this }
				}
				domicile_location.county.holder.liege != root
			}
			domicile_location.county.holder.house.house_head = {
				this != root
				exists = house
				house != root.house
				house = {
					is_powerful_family = yes
				}
				is_available_ai_adult = yes
			}
		}
	}

		government_has_mechanic = administrative
		government_allows = noble_families
		is_available = yes
		house ?= {
			is_powerful_family = yes
		}
		domicile ?= {
			is_domicile_type = mages_commune
			trigger_if = {
				limit = {
					root = { top_liege != this }
				}
				domicile_location.county.holder.liege != root
			}
			domicile_location.county.holder.house.house_head = {
				this != root
				exists = house
				house != root.house
				house = {
					is_powerful_family = yes
				}
				is_available_ai_adult = yes
			}
		}
	}

	immediate = {
		domicile ?= {
			domicile_location.county = {
				save_scope_as = location
				holder.house.house_head = {
					save_scope_as = other_family
				}
			}
		}
	}

	# Have them over for a good dinner
	option = {
		name = mages_commune_events_2.2.a
		change_influence = minor_influence_loss
		duel = {
			skill = diplomacy
			target = scope:other_family
			50 = {
				compare_modifier = {
					value = scope:duel_value
					multiplier = 3.5
					min = -49
				}
				send_interface_toast = {
					title = mages_commune_events_2.2.a.win
					left_icon = root
					right_icon = scope:other_family
					scope:other_family = {
						progress_towards_friend_effect = {
							REASON = friend_dinner
							CHARACTER = root
							OPINION = default_friend_opinion
						}
					}
				}
			}
			50 = {
				compare_modifier = {
					value = scope:duel_value
					multiplier = -3.5
					min = -49
				}
				send_interface_toast = {
					title = mages_commune_events_2.2.a.lose
					left_icon = root
					right_icon = scope:other_family
				}
			}
		}
		stress_and_fulfillment_impact = {
			ambitious = minor_stress_impact_gain
			shy = minor_stress_impact_gain
			vengeful = minor_stress_impact_gain
		}
		ai_chance = {
			base = 10
			opinion_modifier = {
				opinion_target = scope:other_family
				multiplier = 0.1
			}
			ai_value_modifier = {
				ai_sociability = 1.5
				ai_rationality = 1.5
				ai_boldness = -1.5
				ai_vengefulness = -1.5
			}
		}
	}

	# Spread your influence in their lands
	option = {
		name = mages_commune_events_2.2.b
		scope:other_family = {
			progress_towards_rival_effect = {
				REASON = rival_plotting_against_me
				CHARACTER = root
				OPINION = default_rival_opinion
			}
		}
		duel = {
			skills = { intrigue stewardship }
			target = scope:other_family
			50 = {
				compare_modifier = {
					value = scope:duel_value
					multiplier = 3.5
					min = -49
				}
				send_interface_toast = {
					title = mages_commune_events_2.2.b.win
					left_icon = root
					right_icon = scope:other_family
					change_influence = medium_influence_gain
					scope:other_family = {
						change_influence = minor_influence_loss
					}
				}
			}
			50 = {
				compare_modifier = {
					value = scope:duel_value
					multiplier = -3.5
					min = -49
				}
				send_interface_toast = {
					title = mages_commune_events_2.2.b.lose
					left_icon = root
					right_icon = scope:other_family
					change_influence = medium_influence_loss
					scope:other_family = {
						change_influence = minor_influence_gain
					}
				}
			}
		}
		stress_and_fulfillment_impact = {
			compassionate = minor_stress_impact_gain
			gregarious = minor_stress_impact_gain

					}

							house_has_feud_relation_with_trigger = { TARGET = scope:house_head_temp }
						}
					}
				}
			}
		}
