# Magocracy: domicile artifact slots also available in the Mage's Commune. Keep in sync with 00_default.txt

library_1 = {
	icon = artifact_library
	type = "library"
	category = court

	available = {
		domicile ?= {
			OR = {
				is_domicile_type = merchant_company
				is_domicile_type = mages_commune
			}
			custom_tooltip = {
				text = library_1_requirements
				OR = {
					has_domicile_parameter = company_unlock_library_slot_1
					has_domicile_parameter = company_unlock_library_slot_2
					has_domicile_parameter = company_unlock_library_slot_3
				}
			}
		}
	}
}
