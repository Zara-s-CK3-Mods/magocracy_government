## The Conclave of Magisters
#
# Composition decides who sits in the Conclave, authority decides what it must approve,
# and the threshold decides how many votes a proposal needs.
# The head's term of office uses the vanilla republic term length laws (see zz_magocracy_succession_laws.txt).
#
@conclave_law_prestige_cost = 500

### Composition

# The head of the Magocracy and every landed adult magister among their vassals
magocracy_conclave_composition_magisters = {
	law_group_type = republic_assembly_composition_laws
	index = 12

	can_keep = { magocracy_conclave_realm_trigger = yes }
	can_have = { magocracy_conclave_realm_trigger = yes }
	can_pass = { magocracy_conclave_can_pass_law_trigger = yes }
	potential = { government_has_flag = government_is_magocracy }
	should_start_with = { government_has_flag = government_is_magocracy }
	pass_cost = { prestige = @conclave_law_prestige_cost }
	on_after_pass = { update_assembly = yes }

	assembly = {
		major = {
			sources = { vassals liege }
			limit = {
				OR = {
					magocracy_is_conclave_head_trigger = yes
					AND = {
						has_trait = zz_magister
						is_adult = yes
					}
				}
			}
		}
	}

	requires_approve = { magocracy_conclave_requires_approval_trigger = yes }
	ai_will_do = { value = 10 }
}

# A High Conclave of great magisters, with lesser magisters and court magisters as minor members
magocracy_conclave_composition_high = {
	law_group_type = republic_assembly_composition_laws
	index = 14

	can_keep = { magocracy_conclave_realm_trigger = yes }
	can_have = {
		magocracy_conclave_realm_trigger = yes
		highest_held_title_tier >= tier_kingdom
	}
	can_pass = { magocracy_conclave_can_pass_law_trigger = yes }
	potential = { government_has_flag = government_is_magocracy }
	pass_cost = { prestige = @conclave_law_prestige_cost }
	on_after_pass = { update_assembly = yes }

	assembly = {
		major = {
			sources = { vassals liege }
			limit = {
				OR = {
					magocracy_is_conclave_head_trigger = yes
					AND = {
						has_trait = zz_magister
						highest_held_title_tier >= tier_duchy
						is_adult = yes
					}
				}
			}
		}
		minor = {
			sources = { vassals courtiers }
			limit = {
				has_trait = zz_magister
				is_adult = yes
				NOT = { highest_held_title_tier >= tier_duchy }
			}
		}
	}

	requires_approve = { magocracy_conclave_requires_approval_trigger = yes }
	ai_will_do = { value = 5 }
}
