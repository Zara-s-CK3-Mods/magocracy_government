magocracy_arcane_study_decree_decision = {
	title = magocracy_arcane_study_decree_decision
	desc = magocracy_arcane_study_decree_decision_desc
	selection_tooltip = magocracy_arcane_study_decree_decision_tooltip
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_erudition.dds"
	}

	sort_order = 1
	decision_group_type = republic_decree
	is_invisible = yes # Shown in the Republic window

	cooldown = { years = 10 }

	cost = {
		treasury_or_gold = { value = total_republic_decree_gold_cost }
	}

	is_shown = {
		government_has_flag = government_is_magocracy
		has_gdp_dlc_trigger = yes
	}

	is_valid = {
		custom_tooltip = {
			text = you_are_republic_leader_desc
			magocracy_is_conclave_head_trigger = yes
		}
	}

	is_valid_showing_failures_only = {
		is_available_quick = {
			adult = yes
			at_war = no
			imprisoned = no
			incapable = no
		}
	}

	effect = {
		magocracy_list_realm_mages_effect = yes
		every_in_list = {
			list = magocracy_realm_mages
			custom = magocracy_every_realm_mage
			add_character_modifier = {
				modifier = magocracy_arcane_study_modifier
				years = 10
			}
		}
		gdp_consume_republic_assembly_preparation_effect = yes
		gdp_enacted_decree_effect = yes
	}

	requires_approve = { magocracy_decree_requires_approval_trigger = yes }

	ai_approval = {
		value = 0
		if = { # Pledged supporters only show their pledge
			limit = {
				liege ?= {
					this != root
					is_target_in_variable_list = {
						name = decree_approvers
						target = root
					}
				}
			}
			add = {
				value = 100
				desc = decree_pledged_support_desc
			}
		}
		else = {
			add = republic_assembly_preparation_approval_value
			add = republic_caucus_room_approval_value
			if = {
				limit = { has_trait = zz_magister }
				add = {
					value = 10
					desc = magocracy_decree_mage_voter_desc
				}
			}
		}
	}

	ai_check_interval = 120

	ai_potential = {
		this = liege
	}

	ai_will_do = {
		base = 10
	}
}

					target = root

					}
				}
			}
			add = {
				value = 100
				desc = decree_pledged_support_desc
			}
		}
		else = {
			add = republic_assembly_preparation_approval_value
			add = republic_caucus_room_approval_value
			if = {
				limit = { has_trait = zz_magister }
				add = {
					value = -10
					desc = magocracy_decree_mage_voter_desc
				}
			}
		}
	}

	ai_check_interval = 120

	ai_potential = {
		this = liege
	}

		base = 5
	}
