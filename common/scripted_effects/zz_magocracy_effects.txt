# Switch the scoped administrative character to the Magocracy and give their noble family a Mage's Commune
magocracy_change_government_effect = {
	if = {
		limit = { NOT = { has_government = magocracy_government } }
		change_government = magocracy_government
	}
	tgp_domicile_conversion_when_changing_government_type = {
		NEW_DOMICILE_TYPE = mages_commune
		NEW_GOVERNMENT_TYPE = magocracy_government
	}
	magocracy_family_title_conversion_effect = yes
}

# Give the scoped character a Magocracy family title (title_type_magocracy_family), saved as scope:new_title
magocracy_give_family_title_effect = {
	save_scope_as = magocracy_family_founder
	create_dynamic_title = {
		name = MAGOCRACY_FAMILY_TITLE_NAME
		tier = county
		type = title_type_magocracy_family
		holder = scope:magocracy_family_founder
		article = DEFAULT_TITLE_NAME_ARTICLE
		save_scope_as = new_title
	}
	scope:new_title ?= {
		add_title_law = noble_family_succession_law
		set_variable = magocracy_family_title
	}
}

# Swap the scoped character's plain noble family title for a Magocracy family title (so they can own trade posts).
# The new title takes over the old one's history, as vanilla does when a family changes government.
magocracy_family_title_conversion_effect = {
	if = {
		limit = {
			has_government = magocracy_government
			any_held_title = {
				is_noble_family_title = yes
				NOT = { has_variable = magocracy_family_title }
			}
		}
		hidden_effect = {
			random_held_title = {
				limit = {
					is_noble_family_title = yes
					NOT = { has_variable = magocracy_family_title }
				}
				save_scope_as = old_nf_title
			}
			# Keep this order: with no title left the family has no liege, and on_title_destroyed turns an independent admin ruler feudal
			magocracy_give_family_title_effect = yes
			scope:new_title ?= { copy_title_history = scope:old_nf_title }
			destroy_title = scope:old_nf_title
		}
	}
}

# Convert the scoped ruler and their realm into a Magocracy
magocracy_convert_realm_effect = {
	save_scope_as = magocracy_founder
	# Feudal, clan and republic realms first go through the vanilla administrative reform (holdings, noble families, vassals)
	if = {
		limit = { NOT = { government_has_mechanic = administrative } }
		convert_to_administrative_from_feudalism_effect = { GOVERNMENT_TO_ADOPT = flag:administrative }
	}
	magocracy_change_government_effect = yes
	hidden_effect = {
		every_vassal_or_below = {
			limit = {
				government_has_mechanic = administrative
				NOT = { has_government = magocracy_government }
			}
			magocracy_change_government_effect = yes
		}
	}
	custom_tooltip = magocracy_vassals_convert_tt
	if = {
		limit = { is_independent_ruler = yes }
		add_realm_law_skip_effects = magocracy_elective_succession_law
		magocracy_ensure_conclave_laws_effect = yes
		hidden_effect = { update_assembly = yes }
		# Again once the realm laws have been assigned
		trigger_event = { id = zz_magocracy_events.1 days = 1 }
	}
}

# Keep characters of a Magocracy realm on the Magocracy when vanilla scripts hand them plain administrative government
magocracy_sync_government_effect = {
	if = {
		limit = {
			is_ruler = yes
			government_has_mechanic = administrative
			NOT = { has_government = magocracy_government }
			top_liege ?= {
				this != root
				has_government = magocracy_government
			}
			NOR = {
				government_has_flag = government_is_celestial
				government_has_flag = government_is_meritocratic
				government_has_flag = government_is_japan_administrative
				government_has_flag = government_is_steppe_admin
			}
		}
		magocracy_change_government_effect = yes
	}
}

# The engine does not hand a converted realm the Magocracy's republic laws, and without a membership law it seats no Conclave.
# Give the scoped head of the Magocracy the default Magocracy law of each republic group it has no law in.
# Runs every year: a group holding any law is left alone, since a vanilla republic law (such as the Roman Senate) is a valid choice.
magocracy_ensure_conclave_laws_effect = {
	if = {
		limit = {
			NOR = {
				has_realm_law = magocracy_conclave_composition_magisters
				has_realm_law = magocracy_conclave_composition_houses
				has_realm_law = magocracy_conclave_composition_high
				has_realm_law = magocracy_conclave_composition_all_mages
				has_realm_law = republic_assembly_composition_patrician
				has_realm_law = republic_assembly_composition_council
				has_realm_law = republic_assembly_composition_general
				has_realm_law = republic_assembly_composition_parliament
				has_realm_law = republic_assembly_composition_communal
				has_realm_law = republic_assembly_composition_roman_senate
			}
		}
		add_realm_law_skip_effects = magocracy_conclave_composition_magisters
	}
	if = {
		limit = {
			NOR = {
				has_realm_law = magocracy_conclave_authority_constitutional
				has_realm_law = magocracy_conclave_authority_legislative
				has_realm_law = magocracy_conclave_authority_ceremonial
				has_realm_law = republic_assembly_authority_constitutional
				has_realm_law = republic_assembly_authority_legislative
				has_realm_law = republic_assembly_authority_ceremonial
			}
		}
		add_realm_law_skip_effects = magocracy_conclave_authority_legislative
	}
	if = {
		limit = {
			NOR = {
				has_realm_law = magocracy_conclave_voting_thresholds_simple_majority
				has_realm_law = magocracy_conclave_voting_thresholds_supermajority
				has_realm_law = republic_assembly_voting_thresholds_simple_majority
				has_realm_law = republic_assembly_voting_thresholds_large_majority
				has_realm_law = republic_assembly_voting_thresholds_unanimous
			}
		}
		add_realm_law_skip_effects = magocracy_conclave_voting_thresholds_simple_majority
	}
	if = {
		limit = {
			NOR = {
				has_realm_law = magocracy_type_warmage
				has_realm_law = magocracy_type_arcane
				has_realm_law = magocracy_type_merchant
				has_realm_law = republic_type_communal
				has_realm_law = republic_type_civic
				has_realm_law = republic_type_popular
				has_realm_law = republic_type_maritime
				has_realm_law = republic_type_noble
				has_realm_law = republic_type_roman
			}
		}
		add_realm_law_skip_effects = magocracy_type_arcane
	}
	# Term of office for the elected head (vanilla republic term laws, default 10 years)
	if = {
		limit = {
			NOR = {
				has_realm_law = republic_assembly_term_length_1
				has_realm_law = republic_assembly_term_length_2
				has_realm_law = republic_assembly_term_length_3
				has_realm_law = republic_assembly_term_length_4
				has_realm_law = republic_assembly_term_length_5
				has_realm_law = republic_assembly_term_length_indefinite
			}
		}
		add_realm_law_skip_effects = republic_assembly_term_length_2
	}
	# Military budget (vanilla treasury law; see zz_magocracy_vanilla_military_budget_trigger.txt), default 20%
	if = {
		limit = {
			NOR = {
				has_realm_law = budget_allocation_military_0
				has_realm_law = budget_allocation_military_5
				has_realm_law = budget_allocation_military_10
				has_realm_law = budget_allocation_military_15
				has_realm_law = budget_allocation_military_20
				has_realm_law = budget_allocation_military_25
				has_realm_law = budget_allocation_military_30
				has_realm_law = budget_allocation_military_35
				has_realm_law = budget_allocation_military_40
				has_realm_law = budget_allocation_military_45
				has_realm_law = budget_allocation_military_50
				has_realm_law = budget_allocation_military_55
				has_realm_law = budget_allocation_military_60
				has_realm_law = budget_allocation_military_65
				has_realm_law = budget_allocation_military_70
				has_realm_law = budget_allocation_military_75
				has_realm_law = budget_allocation_military_80
				has_realm_law = budget_allocation_military_85
				has_realm_law = budget_allocation_military_90

# Interaction scope: only the patrician being appointed as a governor joins the Magocracy, other republic rulers are left alone
magocracy_appoint_patrician_governor_effect = {
	if = {
		limit = {
			scope:puppet_or_actor = { government_has_flag = government_is_magocracy }
			scope:recipient = {
				government_has_flag = government_is_republic
				NOT = { has_government = magocracy_government }
			}
		}
		custom_tooltip = magocracy_appoint_patrician_governor_tt
		hidden_effect = {
			scope:recipient = {
				magocracy_change_government_effect = yes
			}
		}
	}
}
